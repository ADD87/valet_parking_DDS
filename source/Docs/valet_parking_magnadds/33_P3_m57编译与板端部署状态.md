  -v "$(pwd):/workspace" \
  elc-artifactory-hoo.magna.global:443/84244-cn-sw-docker/l5i4550_m57_mcm6c_jammy_env:latest

./build_app.sh --build=m57 --mode=release

## 板端部署步骤

1. 将 `.tar.gz` 拷贝到 m57 板
2. 解压：`tar xzf valet_parking_m57_runtime_*.tar.gz`
3. 设置库路径：`export LD_LIBRARY_PATH=$PWD/lib:$LD_LIBRARY_PATH`
4. 启动 runner：`./app/valet_parking_runner`
5. 验证 DDS discovery 和 topic 收发
